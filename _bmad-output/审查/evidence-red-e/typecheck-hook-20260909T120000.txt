### lefthook python-typecheck 预演 — 模拟 hook 对本卡 staged 文件跑 pyright
-- lefthook.yml python-typecheck 块 --
    python-typecheck:
      glob: "backend/app/*.py"
      run: |
        # Activate backend venv for pyright
        if [ -f "backend/.venv/bin/activate" ]; then source backend/.venv/bin/activate; fi
        # Resolve the binary explicitly: venv first, then PATH. Empty == not installed.
        if [ -x "backend/.venv/bin/pyright" ]; then
          PYRIGHT_BIN="backend/.venv/bin/pyright"
        elif command -v pyright >/dev/null 2>&1; then
          PYRIGHT_BIN="$(command -v pyright)"
        else
          PYRIGHT_BIN=""
        fi
        if [ -z "$PYRIGHT_BIN" ]; then
          echo "[Python] SKIP: pyright not installed (looked for backend/.venv/bin/pyright, then PATH)."
          echo "[Python] SKIP: typecheck did NOT run -- this is NOT a pass. See pyrightconfig.json."
          exit 0
        fi
        echo "[Python] Running pyright type check ($PYRIGHT_BIN)..."
        "$PYRIGHT_BIN" {staged_files} 2>&1
        PYRIGHT_EXIT=$?
        if [ $PYRIGHT_EXIT -ne 0 ]; then
          echo "[Python] Type errors found! Review and fix before committing."
          echo "[Python] Hint: Some errors may be pre-existing. Focus on YOUR changes."
        fi
        echo "[Python] Typecheck done (exit: $PYRIGHT_EXIT)."
        exit $PYRIGHT_EXIT
