def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def run_tests() -> tuple[set[str], bool]:
    """返回 (失败的测试函数名集合, 是否发生了收集期错误)。

    **同时收 FAILED 与 ERROR**: 变异体若让模块 import 崩掉, pytest 报的是 ERROR,
    只收 FAILED 会把「门根本没跑」读成「一条都没红 = 门不承重」= 假 SURVIVED。
    """
    proc = subprocess.run(
        [str(PYTEST), "-q", "-p", "no:cacheprovider", "--tb=no", "-rfE", TEST_FILE],
        cwd=WT / "backend",
        capture_output=True,
        text=True,
        env={"PATH": "/usr/bin:/bin", "PYTHONDONTWRITEBYTECODE": "1", "HOME": str(Path.home())},
    )
    failed: set[str] = set()
    collect_error = False
    for line in proc.stdout.splitlines():
        m = re.match(r"^FAILED .*::(\w+)", line)
        if m:
            failed.add(m.group(1))
            continue
        if line.startswith("ERROR "):
            collect_error = True
            m = re.match(r"^ERROR .*::(\w+)", line)
            if m:
                failed.add(m.group(1))
    return failed, collect_error


def compiles(text: str) -> bool:
    """变异体语法自检 —— 语法错的变异体不是「有效变异」, 单列一种裁决。"""
    try:
        compile(text, str(VERIFIER), "exec")
        return True
    except SyntaxError:
        return False


def main() -> int:
    baseline_sha = sha(VERIFIER)
    original = VERIFIER.read_text(encoding="utf-8")
    print(f"# baseline sha  : {baseline_sha}")

    green, green_err = run_tests()
    print(f"# 未变异时失败集: {sorted(green) or '(空 — 全绿)'}  收集期错误={green_err}")
    verdicts = []
    if green or green_err:
        print("!! 基线非全绿, 变异结论不可信")
        return 1

    for name, old, new, expected in MUTATIONS:
        assert original.count(old) == 1, f"{name}: 锚点不唯一 ({original.count(old)} 次)"
        mutated = original.replace(old, new)
        # 语法自检: 语法错的变异体等于「门没机会跑」, 既不能记 KILLED 也不能记 SURVIVED
        if not compiles(mutated):
            verdicts.append((name, False, set(), expected, True))
            print(f"\n## {name}: SYNTAX-INVALID — 变异体语法非法, 本条不成立(须改写)")
            continue
        VERIFIER.write_text(mutated, encoding="utf-8")
        try:
            failed, collect_error = run_tests()
        finally:
            VERIFIER.write_text(original, encoding="utf-8")  # 无条件还原
        restored = sha(VERIFIER) == baseline_sha
        ok = failed == expected and restored and not collect_error
        verdicts.append((name, ok, failed, expected, restored))
        status = "KILLED-EXACT" if ok else "MISMATCH"
        print(f"\n## {name}: {status}  (还原={'ok' if restored else 'FAILED'})")
        print(f"   期望红: {sorted(expected)}")
        print(f"   实际红: {sorted(failed)}")
        if collect_error:
            print("   ⚠️ 出现收集期 ERROR — 门没跑起来, 不能当作承重证据")
        if failed - expected:
            print(f"   ⚠️ 连坐 (本不该红): {sorted(failed - expected)}")
        if expected - failed:
            print(f"   ⚠️ 未被杀 (门不承重): {sorted(expected - failed)}")

    final_ok = sha(VERIFIER) == baseline_sha
    print(f"\n# 结束 sha 复核 : {'逐字节相同' if final_ok else '❌ 未还原'}")
    all_ok = all(v[1] for v in verdicts) and final_ok
    print(f"# 总判定        : {'PASS — 四类各自独立承重' if all_ok else 'FAIL'}")
    return 0 if all_ok else 1


if __name__ == "__main__":
    sys.exit(main())
