   873	def _extract_script(html: str) -> str:
   874	    """从实际响应体取出**整个** <script> 原文 — 且必须与浏览器 tokenizer 同源。
   875	
   876	    HTML 解析对 script 结束符**大小写不敏感**、正文里出现注释开合也会改变
   877	    解析状态 (round-2 HIGH-3 实证: `// </SCRIPT>` 后的"好代码"能骗过大小写
   878	    敏感的正则, 但浏览器根本不会执行它)。所以除定位外, 还要拒绝一切会让
   879	    「正则提取的字节」与「浏览器执行的字节」分叉的序列。
   880	    """
   881	    # 开标签: 大小写不敏感地只允许一个, 且必须恰是字面 <script> (无属性)
   882	    openings = re.findall(r"<script\b[^>]*>", html, flags=re.I)
   883	    assert openings == ["<script>"], f"script 开标签形态变了: {openings!r}"
   884	    _prefix, sep, rest = html.partition("<script>")
   885	    assert sep, "script 开标签未命中"
   886	    # 结束符**总数**恰为 1 (大小写不敏感) — 正文任何位置多出结束符 (哪怕在
   887	    # JS 注释里), 浏览器都会在那里提前终止脚本; maxsplit 截断式提取会把它
   888	    # 当切割点吞掉, 那道门就是死的 (round-3 自查), 数段数才漏不掉。
   889	    # round-3 HIGH-3: 浏览器结束标签语言还接受 solidus 分隔 (`</script/`) 与
   890	    # 携带属性 (`</script x=y>`) — `</script\s*>` 数不到这些形态。
   891	    # round-4 HIGH-3 补口: 前缀后紧跟 ASCII 字母 (`</scriptx`) 不是结束标签,
   892	    # 浏览器不在此闭合 — 计数与切割都带 lookahead 对齐 tokenizer, 否则合法
   893	    # 页面会被假截断, 截断后的字节与浏览器执行的字节分叉
   894	    hits = re.findall(r"</script(?=[\t\n\f\r />])", rest, flags=re.I)
   895	    assert len(hits) == 1, f"script 结束标签出现 {len(hits)} 次 (应恰 1) — 浏览器会在正文处提前终止"
   896	    src = re.split(r"</script(?=[\t\n\f\r />])", rest, maxsplit=1, flags=re.I)[0]
   897	    # 正文分叉面: HTML 注释开合改变 tokenizer 状态 (script data escaped)

   899	    assert "__URLS_JSON__" not in src, "占位符没被替换 — 页面发出去的是模板本身"
   900	    return src
