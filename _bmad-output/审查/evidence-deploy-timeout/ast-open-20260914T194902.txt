import ast
p="backend/tests/unit/test_deploy_vault_sh.py"
t=ast.parse(open(p).read()); w=[]; wo=[]
for n in ast.walk(t):
    if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=="run" \
       and getattr(n.func.value,"id","")=="subprocess":
        (w if any(k.arg=="timeout" for k in n.keywords) else wo).append(n.lineno)
print("total",len(w)+len(wo),"with",len(w),sorted(w),"WITHOUT",len(wo),sorted(wo))
