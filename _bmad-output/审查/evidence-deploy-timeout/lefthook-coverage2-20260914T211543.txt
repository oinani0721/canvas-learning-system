--- 驱动源码 ---
#!/usr/bin/env bash
set -uo pipefail
ROOT="$1"; SP="$2"; cd "$ROOT"
PY=backend/tests/unit/test_deploy_vault_sh.py
SNAP="$SP/lefthook-probe-snapshot.py"
cp "$PY" "$SNAP"
trap 'cp "$SNAP" "$ROOT/$PY"; git -C "$ROOT" reset -q; echo "[EXIT trap] 已还原并清暂存区"' EXIT
echo "== 跑前 shasum =="; shasum -a 256 "$PY"
printf '\n# lefthook 覆盖面探针（临时行，跑完无条件还原）\n' >> "$PY"
git add -- "$PY" scripts/deploy-vault.sh
echo "== 暂存面 =="; git diff --cached --name-only -z | tr '\0' '\n' | sed 's/^/   /'
echo "== lefthook run pre-commit（逐 job 的命中/跳过理由 = 自带锚）=="
/opt/homebrew/bin/lefthook run pre-commit 2>&1 | grep -E -e '❯|✔️|✖|skip|summary' | sed 's/^/   /'
cp "$SNAP" "$PY"; git reset -q
echo "== 还原后 shasum（须逐字同）=="; shasum -a 256 "$PY"
echo "== 还原后工作树 =="; git status --porcelain | grep -v 'lefthook-coverage' | wc -l | tr -d ' ' | sed 's/^/   非本次存档的脏项=/'

[结论] A 全绿 / B 被 format 判要重排 ⇒ 那个 exit status 1 **由探针行引入**，与本卡真实改动无关。
[旁证] 本卡 4 条 commit 全部成功落地，而 pre-commit 非零会中止 commit（commitlint 那次即实测被拦）
       ⇒ python-lint 在每次真实提交时都跑过且通过。
