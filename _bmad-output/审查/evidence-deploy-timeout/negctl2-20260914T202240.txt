#!/usr/bin/env bash
# 负控第四段：把 Codex r1 的三处修复逐个变异回旧行为，验证**指定的那道门**会变红。
set -uo pipefail
ROOT="$1"; SP="$2"
cd "$ROOT"
SH_F=scripts/deploy-vault.sh
SNAP="$SP/negctl2-deploy-vault.sh"
cp "$SH_F" "$SNAP"
trap 'cp "$SNAP" "$ROOT/$SH_F"; echo "[EXIT trap] 已无条件还原 $SH_F"' EXIT
PYTEST="$ROOT/backend/.venv/bin/pytest"

echo "===== 跑前 shasum ====="; shasum -a 256 "$SH_F" backend/tests/unit/test_deploy_vault_sh.py

run_one () {   # $1=名字 $2=nodeid 表达式 $3=期望(RED/GREEN)
  local out rc
  out=$( cd backend && PYTHONDONTWRITEBYTECODE=1 "$PYTEST" -q -p no:cacheprovider \
          tests/unit/test_deploy_vault_sh.py -k "$2" 2>&1 | tail -3 )
  rc=$?
  printf '  [%s] -k %s => %s\n' "$1" "$2" "$(printf '%s' "$out" | tr '\n' ' ')"
}

mutate () { python3 - "$SH_F" "$1" <<'PY'
import sys
p, which = sys.argv[1], sys.argv[2]
s = open(p, encoding="utf-8").read()
if which == "m1":      # 撤回取值范围校验 ⇒ 0 / 000 / 超界重新被放行
    i = s.index('        _cap="${_cap#')
    j = s.index("        # ⛔ 超时信号走", i)
    s = s[:i] + s[j:]
elif which == "m2":    # 超时信号改回「看退出码 124」⇒ 与 npm 自己 exit 124 撞车
    s = s.replace('        case "$_build_out" in\n            *"$_mark"*)', '        case "$_build_rc" in\n            124)')
elif which == "m3":    # 删掉离线开关注入
    s = s.replace('               npm_config_offline="$CLS_NPM_BUILD_OFFLINE" \\\n', '')
else:
    raise SystemExit("unknown mutation")
open(p, "w", encoding="utf-8").write(s)
print(f"  [变异 {which}] 已应用，bash -n:", end=" ")
PY
bash -n "$SH_F" && echo "rc=0" || echo "rc=1 ⇒ 变异体语法坏了, 本段作废"
}

for M in m1 m2 m3; do
  echo; echo "----- 变异 $M -----"
  cp "$SNAP" "$SH_F"
  mutate "$M"
  case "$M" in
    m1) run_one "$M 指定门(取值校验)" 'cap_values' "RED"; run_one "$M 旁证(挂起承重)" 'is_walltime_capped' "GREEN" ;;
    m2) run_one "$M 指定门(rc124 不得报超时)" 'not_reported_as_timeout' "RED" ;;
    m3) run_one "$M 指定门(离线开关到位)" 'does_not_kill_a_fast_build' "RED" ;;
  esac
done

cp "$SNAP" "$SH_F"
echo; echo "===== 还原后 shasum（须与跑前逐字同）====="; shasum -a 256 "$SH_F" backend/tests/unit/test_deploy_vault_sh.py
echo "===== 还原后指定门应全绿 ====="
run_one "还原" 'cap_values or not_reported_as_timeout or does_not_kill_a_fast_build' "GREEN"
