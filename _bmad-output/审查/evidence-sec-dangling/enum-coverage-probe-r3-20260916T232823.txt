print("### E 组 —— 真实快照（改枚举面不改本仓结论）")
real = json.load(open(ROOT / "backend/openapi.json", encoding="utf-8"))
got_real = list(enumerate_refs(real))
per_op = [g for g in got_real if g[0] != "<root>"]
declared = set((real.get("components") or {}).get("securitySchemes") or {})
dangling = [g for g in got_real if g[1] not in declared]
sysn = [g for g in dangling if "/system/" in g[0]]
ok_e = (len(got_real) == 32 and len(per_op) == 31 and declared == {"InternalApiKey"}
        and dangling == [] and sysn == [])
RESULTS.append(ok_e)
DISCRIMINATING.append(None)   # E1 不是加固用例, 是「改枚举面不改本仓结论」的回归锁
print(f"  [{'PASS' if ok_e else 'FAIL'}] E1 真实快照 = 引用 32 / per-op 31 / root 1 / 声明集 {{InternalApiKey}} / 悬空 0")
print(f"        实际: 引用={len(got_real)} per-op={len(per_op)} root={len(got_real)-len(per_op)} "
      f"声明集={sorted(declared)} 悬空={len(dangling)} 其中/system/*={len(sysn)}")

print()
# ⛔ 计数一律算出来, 不写死（写死过一次: 曾写「13 / 共 20, 其余 5 条」, 13+5≠20）
_t = sum(1 for d in DISCRIMINATING if d is True)
_f = sum(1 for d in DISCRIMINATING if d is False)
_n = sum(1 for d in DISCRIMINATING if d is None)
assert _t + _f + _n == len(RESULTS) == len(DISCRIMINATING), "计数自洽性断言失败"
print(f"ALL_PASS = {all(RESULTS)}  ({sum(RESULTS)}/{len(RESULTS)})")
print(f"分类: 能区分本卡加固 = {_t} 条 | 覆盖性(加固前后都过) = {_f} 条 | 回归锁(真实快照) = {_n} 条"
      f" | 合计 {_t}+{_f}+{_n} = {len(RESULTS)}")
sys.exit(0 if all(RESULTS) else 1)
