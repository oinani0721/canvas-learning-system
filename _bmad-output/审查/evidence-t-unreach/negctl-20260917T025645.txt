def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def handler_order(path: Path, fn_name: str) -> list[str]:
    fn = next(n for n in ast.walk(ast.parse(path.read_text(encoding="utf-8")))
              if getattr(n, "name", "") == fn_name)
    for tr in ast.walk(fn):
        if isinstance(tr, ast.Try):
            order = [getattr(h.type, "attr", getattr(h.type, "id", None)) for h in tr.handlers]
            if "ValidationError" in order and "ValueError" in order:
                return order
    raise AssertionError(f"{path}::{fn_name} 未找到目标 try")


def run_tests() -> tuple[set[str], int]:
    proc = subprocess.run(
        [str(PYTEST), "-q", "-p", "no:cacheprovider", "--no-header", "-rf", TESTFILE],
        cwd=ROOT / "backend", capture_output=True, text=True,
        env={"PYTHONDONTWRITEBYTECODE": "1", "PATH": "/usr/bin:/bin:/usr/sbin:/sbin", "HOME": str(Path.home())},
    )
    out = proc.stdout + proc.stderr
    failed = set()
    for line in out.splitlines():
        if line.startswith("FAILED "):
            nodeid = line.split(" ", 1)[1].split(" - ")[0]
            failed.add(nodeid.split("::", 1)[1] if "::" in nodeid else nodeid)
    tail = [ln for ln in out.splitlines() if " passed" in ln or " failed" in ln]
    print("    pytest 汇总:", tail[-1] if tail else "(无汇总行)")
    return failed, proc.returncode
