"""CARD-R-SLO 负控 3（配对演示）输入生成器。
用途：阶段 A/B 各生成一份 J08 注入件；两阶段唯一差异 = not_measured 项的 meets（A=false, B=true）。
其余字段（含 result=partial）逐字相同，pre-assert 强制。"""
import io, json, re, sys
import yaml

phase = sys.argv[1]
out_path = sys.argv[2]
assert phase in ("A", "B"), phase

y = yaml.safe_load(open("docs/release-evidence/slo-manifest.yaml", encoding="utf-8"))
items = []
for m in y["metrics"]:
    th = m["threshold"]
    if th["locked"] is not None:
        thr = th["locked"]
    elif th["candidate"] is not None:
        thr = f'{th["candidate"]} (candidate)'
    else:
        thr = "(未定)"
    ms = m["measured"]
    if ms["status"] == "measured":
        measured = f"p95={ms['p95']}{m['unit']} (n={ms['n']})"
        cap = re.search(r"≤\s*([0-9.]+)", str(m["threshold"]["candidate"]))
        meets = bool(cap) and float(ms["p95"]) <= float(cap.group(1))
    else:
        measured = "not_measured"
        meets = (phase == "B")   # <<< 唯一变量
    items.append({"metric": m["metric"], "threshold": thr, "measured": measured,
                  "method": m["method"]["command"], "meets": meets})

j = json.load(open("docs/release-evidence/example-backfill-d5/journeys/J08/manifest.json", encoding="utf-8"))
j["slo"] = {"manifest_revision": y["revision"], "measurements": items}
assert j["result"] == "partial", j["result"]  # pre-assert：result 不变量
json.dump(j, open(out_path, "w", encoding="utf-8"), ensure_ascii=False, indent=2)
print(f"pre-assert OK: result={j['result']} (未改)；wrote {out_path} (phase {phase})")
