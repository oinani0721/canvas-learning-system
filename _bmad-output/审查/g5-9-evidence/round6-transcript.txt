   397	def _rollback_published(
   398	    name: str,
   399	    dir_fd: int,
   400	    identity: tuple[int, int] | None = None,
   401	    *,
   402	    expect_sha: str | None = None,
   403	) -> tuple[str, str | None]:
   404	    """撤销**我们自己**刚发布的 target —— 仅当它当前确实还是我们那一份。
   405	
   406	    round-3 HIGH-3: 原实现按 basename 直接 unlink 并静默吞掉失败, 两个问题:
   407	      · identity 快照可能已过时 —— 期间路径被换入他人文件时会误删;
   408	      · 删除失败被吞 ⇒ 错误字节的 target 留在 vault 里, 回执却只报失败。

   411	      · ``identity``  —— 比 (dev, ino), 用于「刚 link 出来、inode 已知」的场景;
   412	      · ``expect_sha`` —— 比内容 SHA, 用于 HIGH-1 那种「目录整体被移走」的场景
   413	        (此时 inode 仍是我们的, 但我们想确认删的确是自己写的那份字节)。
   414	    两者都给则必须同时满足。任一不符 ⇒ **不删**(那不是我们的东西)。
   415	    ⛔ round-4 HIGH-3: 原先只返回 `None | str`, 而 `None` 同时表示四种截然不同的
   416	    结果(已删 / 本就不存在 / identity 不符**没删** / SHA 不符**没删**)。调用方无法
   417	    分辨, 于是一律拼「已撤销该文件」—— **回执与实际副作用直接不一致**。
   418	    改为四态: ("deleted"|"deleted_unsynced"|"absent"|"kept", 说明|None)。
   419	    · deleted = 确实删了**且目录项已持久化**; · deleted_unsynced = 删了但持久化未确认
   420	    (round-5 HIGH-2b: 崩溃后目标可能重现, 调用方不得声称「已撤销」而不加限定);
   421	    · absent = 本就不在; · kept = **判据不符, 故意没删**。
   422	    """
   423	    if identity is None and expect_sha is None:
   424	        # round-4 LOW-1: 两个判据都不给 ⇒ 无从证明「这是我们的」, 拒绝删除。
   425	        return "kept", "未提供任何身份判据, 拒绝删除"
