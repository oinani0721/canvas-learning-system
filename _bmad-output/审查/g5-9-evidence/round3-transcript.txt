   295	
   296	    主 session 并行复核 HIGH-4 的正解。三重校验:
   297	    1. ``O_DIRECTORY|O_NOFOLLOW`` —— 目录本身若是 symlink 直接 ELOOP 失败;

   301	       挂载点的链接时, 即便前两条侥幸通过也会在这里被拦。

   304	    exam = vault / EXAM_DIR
   305	    try:
   306	        dfd = os.open(exam, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
   307	    except OSError as e:
   308	        return (
   309	            None,
   310	            f"{EXAM_DIR}/ 打开失败 (被 symlink 布防或不是目录): {type(e).__name__}",
   311	        )

   313	        st_dfd = os.fstat(dfd)

   315	        st_vault = os.stat(vault)

   318	        return None, f"{EXAM_DIR}/ 身份校验失败: {type(e).__name__}"

   320	        os.close(dfd)
   321	        return None, f"{EXAM_DIR}/ 在打开前后被替换 (inode 变化), 拒绝写入"

   323	        os.close(dfd)
   324	        return None, f"{EXAM_DIR}/ 与 vault 不在同一文件系统 (疑似越界挂载), 拒绝写入"

   328	def _atomic_write(tmp: Path, target: Path, content: str, dir_fd: int) -> str | None:
   329	    """O_EXCL|O_NOFOLLOW 写 tmp → fsync → **link 到 target (no-replace)** →
   330	    unlink tmp → fsync 父目录。
   331	
   332	    W4 纵深防御: O_NOFOLLOW 让 tmp 若是 symlink 直接 ELOOP 失败 (不跟随),
   333	    O_CREAT|O_EXCL 让 tmp 已存在时直接失败 (不覆盖并发/残留文件) —
   334	    lstat 预检与实际写之间的 TOCTOU 窗口由内核标志封死。
   335	    round-3 H6: 落盘不再用 ``os.replace`` (它会**覆盖**预检之后才出现的
   336	    target) —— 改用 ``os.link``: 目标已存在时内核直接 EEXIST, 原子且永不
   337	    覆盖; 成功后 unlink tmp, 目标是同一 inode 的硬链接 (内容已 fsync)。
   338	    round-3 M5: 补 fsync 父目录; tmp 清理失败如实回报而非静默吞掉。
   339	    → (失败原因 | None, 告警 | None)。失败时目标未创建; 告警只表示 tmp
   340	    未清理 (目标已正确落盘), 由调用方如实写进回执而非当作失败。
   341	
   342	    ⛔ 主 session 并行复核 HIGH-4 (实测反例: probe 后把 检验白板/ 换成指向
   343	    vault 外的目录 symlink, create 仍返回 created:true 且**文件落在 vault 外**):
   344	    此前 `_prepare` 的目录守卫、`_symlink_probe` 与本函数的 open/link 全部**按路径**
   345	    操作 —— 检查与使用之间隔着可被替换的路径解析, O_NOFOLLOW 只护住最后一段,
   346	    中段/父目录被换掉时护不住。修法 = **dirfd 锚定**: 调用方用
   347	    O_DIRECTORY|O_NOFOLLOW 打开 检验白板/ 拿到 dfd 并校验它确在 vault 内,
   348	    此后所有操作只用 **basename + dir_fd=dfd**。dfd 钉死的是那一个目录 inode,
   349	    路径事后怎么换都改变不了操作落点 —— 窗口从根本上消失, 而不是被压小。
   350	
   351	    ⚠️ **作用域边界（独立复核明确要求不得外推）**: 本函数保证的是
   352	    「**锚定成功之后**, `_atomic_write` 内部的落点不再依赖路径解析」。
   353	    它**不**等于「整个程序从此完全不解析路径」——
   354	      · `_prepare` 在锚定**之前**仍有 mkdir、`_symlink_probe` 与打开目录时的路径解析;
   355	      · `cmd_undo` 是另一套**按路径**的流程, 不走本锚定。
   356	    写文档/验收单时请照此口径, 不要写成「全程 dirfd 化」。
   357	    """
   358	    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW
   359	    tmp_name, target_name = tmp.name, target.name
   360	    try:
   361	        fd = os.open(tmp_name, flags, 0o644, dir_fd=dir_fd)
   362	    except OSError as e:
   363	        return (
   364	            f"临时文件创建失败 (symlink 布防或残留 {tmp.name}): {type(e).__name__}",
   365	            None,
   366	        )
   367	    try:

   369	            f.write(content)
   370	            f.flush()

   372	        # round-4 H6: 发布的字节必须是**我们刚写的那个 inode** — 关闭 fd 后
   373	        # 按路径 link, 期间 tmp 若被替换会发布他人内容。这里用写入时的 fd 记下
   374	        # (dev,ino), link 之后立刻核对 target 是同一 inode, 不符则撤销发布。

   381	        # codex round-1 HIGH-2 (a): 只比 (dev,ino) 不足 —— 别的进程**原地改写

   383	        # 他人字节 (复核者隔离注入实测: 回执 SHA e51ca99e… 与目标实际 43cb09e0…
   384	        # 分叉, 而 _atomic_write 返回 err=None)。修法 = 发布后**重读目标字节**

   386	        want_sha = hashlib.sha256(content.encode("utf-8")).hexdigest()
   387	        vfd = os.open(target_name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=dir_fd)
   388	        try:
   389	            st_pub = os.fstat(vfd)
   390	            got = b""
   391	            while chunk := os.read(vfd, 1 << 20):
   392	                got += chunk
   393	        finally:
   394	            os.close(vfd)
   395	        same_inode = (st_pub.st_dev, st_pub.st_ino) == (

   401	            # 本分支恰恰说明 target 已**不是我们的 inode**(被并发替换), 按路径
   402	            # 删它等于删掉别人刚创建的文件, 且回执还报失败 → 文件静默丢失。
   403	            # 修法: inode 不符时**绝不删**, 只如实回报; 仅当 inode 仍是我们的
   404	            # (纯字节被原地改写) 才撤销自己的发布。

   411	            raise OSError(
   412	                "published inode mismatch (concurrent replacement; 未删除该文件)"
   413	            )
   414	    except OSError as e:
   415	        if fd >= 0:
   416	            try:
   417	                os.close(fd)
   418	            except OSError:
   419	                pass
   420	        try:
   421	            os.unlink(tmp_name, dir_fd=dir_fd)
   422	        except OSError:
   423	            pass
   424	        kind = (
   425	            "目标已存在 (并发创建), 拒绝覆盖"
   426	            if isinstance(e, FileExistsError)
   427	            else "原子写失败"
   428	        )
   429	        return f"{kind}: {type(e).__name__}", None
   430	    warn = None
   431	    try:
   432	        os.unlink(tmp_name, dir_fd=dir_fd)
   433	    except OSError as e:
   434	        warn = f"临时文件未能清理 ({tmp.name}: {type(e).__name__})，下次 create 同 ts 会被拒绝，请手动删除"
   435	    try:
   436	        os.fsync(dir_fd)  # dirfd 已在手, 直接 fsync 它 (不再按路径重开)
   437	    except OSError:
   438	        pass
   439	    return None, warn
